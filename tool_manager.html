Tool Integration API

Use our API to integrate tools programmatically:

# Python example to execute a tool via API
import requests
import json

api_url = "http://localhost:5000/api/execute-tool"
headers = {'Content-Type': 'application/json'}

# Example payload for Wireshark analysis
payload = {
    "tool": "wireshark",
    "params": {
        "file": "capture.pcap",
        "filter": "http contains otp"
    }
}

response = requests.post(api_url, headers=headers, data=json.dumps(payload))
results = response.json()

print(f"Analysis complete: {len(results['potential_otps'])} OTPs found")
